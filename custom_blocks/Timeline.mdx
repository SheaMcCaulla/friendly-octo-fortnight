import React from 'react';

export const TIMELINE_TYPE_STYLES = {
  added: 'bg-green-500 text-white',
  changed: 'bg-blue-500 text-white',
  fixed: 'bg-amber-500 text-white',
  deprecated: 'bg-orange-500 text-white',
  removed: 'bg-red-500 text-white',
  security: 'bg-purple-500 text-white',
};

export const TimelineItem = ({ date, version, type, title, children, isLast = false }) => {
  const badgeClass = TIMELINE_TYPE_STYLES[type?.toLowerCase()] || 'bg-gray-400 text-white';

  return (
    <li className="relative pb-8 pl-8 last:pb-0">
      {!isLast && (
        <span
          className="absolute bottom-0 left-[7px] top-4 w-px bg-gray-200 dark:bg-gray-700"
          aria-hidden="true"
        />
      )}
      <span
        className="absolute left-0 top-1.5 h-3.5 w-3.5 rounded-full border-2 border-white bg-gray-400 dark:border-gray-900"
        aria-hidden="true"
      />
      <div className="flex flex-wrap items-center gap-2">
        {version && <span className="text-sm font-semibold text-gray-900 dark:text-white">{version}</span>}
        {type && (
          <span className={`rounded-full px-2 py-0.5 text-xs font-medium capitalize ${badgeClass}`}>{type}</span>
        )}
        {date && <time className="text-xs text-gray-500 dark:text-gray-400">{date}</time>}
      </div>
      {title && <p className="mb-1 mt-1 font-medium text-gray-800 dark:text-gray-100">{title}</p>}
      <div className="text-sm text-gray-600 dark:text-gray-300">{children}</div>
    </li>
  );
};

export const Timeline = ({ children }) => {
  const items = React.Children.toArray(children);

  return (
    <ol className="m-0 list-none p-0">
      {items.map((child, index) =>
        React.isValidElement(child) ? React.cloneElement(child, { key: index, isLast: index === items.length - 1 }) : child,
      )}
    </ol>
  );
};

    Added a new `POST /exports` endpoint for exporting large datasets asynchronously.
  </TimelineItem>
  <TimelineItem version="v2.2.1" type="fixed" date="June 18, 2026" title="Pagination bug">
    Fixed an issue where `cursor` tokens could expire before pagination completed.
  </TimelineItem>
  <TimelineItem version="v2.2.0" type="deprecated" date="June 2, 2026" title="Legacy auth header">
    The `X-Api-Token` header is deprecated in favor of `Authorization: Bearer`.
  </TimelineItem>
</Timeline>
