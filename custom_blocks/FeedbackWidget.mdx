name: FeedbackWidget
---

import React, { useState } from 'react';

export const FeedbackWidget = ({
  prompt = 'Was this page helpful?',
  yesLabel = 'Yes',
  noLabel = 'No',
  webhookUrl,
  onSubmit,
}) => {
  const [status, setStatus] = useState('idle');
  const [helpful, setHelpful] = useState(null);
  const [comment, setComment] = useState('');

  const sendFeedback = async (payload) => {
    setStatus('submitting');

    try {
      if (webhookUrl) {
        await fetch(webhookUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            ...payload,
            page: typeof window !== 'undefined' ? window.location.href : undefined,
          }),
        });
      }
      if (onSubmit) onSubmit(payload);
    } catch (error) {
      console.error('Failed to send feedback:', error);
    } finally {
      setStatus('submitted');
    }
  };

  const handleChoice = (choice) => {
    setHelpful(choice);
    if (choice) {
      sendFeedback({ helpful: choice, comment: '' });
    } else {
      setStatus('commenting');
    }
  };

  const handleCommentSubmit = (event) => {
    event.preventDefault();
    sendFeedback({ helpful, comment });
  };

  if (status === 'submitted') {
    return (
      <div
        role="status"
        className="rounded-md border border-gray-200 bg-white p-4 text-sm text-gray-700 dark:border-gray-700 dark:bg-gray-800 dark:text-gray-200"
      >
        <i className="fa-solid fa-circle-check mr-2 text-green-500" aria-hidden="true" />
        Thanks for your feedback!
      </div>
    );
  }

  return (
    <div className="rounded-md border border-gray-200 bg-white p-4 dark:border-gray-700 dark:bg-gray-800">
      <p className="m-0 mb-3 text-sm font-semibold text-gray-800 dark:text-white">{prompt}</p>
      {status !== 'commenting' ? (
        <div className="flex gap-2">
          <button
            type="button"
            onClick={() => handleChoice(true)}
            disabled={status === 'submitting'}
            className="flex items-center gap-2 rounded-md border border-gray-300 px-3 py-1.5 text-sm text-gray-700 hover:bg-gray-100 disabled:opacity-60 dark:border-gray-600 dark:text-gray-200 dark:hover:bg-gray-700"
          >
            <i className="fa-solid fa-thumbs-up" aria-hidden="true" />
            {yesLabel}
          </button>
          <button
            type="button"
            onClick={() => handleChoice(false)}
            disabled={status === 'submitting'}
            className="flex items-center gap-2 rounded-md border border-gray-300 px-3 py-1.5 text-sm text-gray-700 hover:bg-gray-100 disabled:opacity-60 dark:border-gray-600 dark:text-gray-200 dark:hover:bg-gray-700"
          >
            <i className="fa-solid fa-thumbs-down" aria-hidden="true" />
            {noLabel}
          </button>
        </div>
      ) : (
        <form onSubmit={handleCommentSubmit}>
          <label htmlFor="feedback-comment" className="mb-1 block text-xs text-gray-600 dark:text-gray-300">
            What could be better? (optional)
          </label>
          <textarea
            id="feedback-comment"
            rows={3}
            value={comment}
            onChange={(event) => setComment(event.target.value)}
            className="mb-2 w-full rounded-md border border-gray-300 p-2 text-sm text-gray-800 dark:border-gray-600 dark:bg-gray-900 dark:text-gray-100"
          />
          <button
            type="submit"
            disabled={status === 'submitting'}
            className="rounded-md bg-blue-500 px-3 py-1.5 text-sm font-medium text-white hover:bg-blue-600 disabled:opacity-60"
          >
            {status === 'submitting' ? 'Sending…' : 'Send feedback'}
          </button>
        </form>
      )}
    </div>
  );
};

<FeedbackWidget prompt="Was this page helpful?" webhookUrl="https://example.com/api/doc-feedback" />
